# The board is recognised in either state. The bootloader and the sketch
# enumerate with different product IDs so tools can tell them apart, and both
# are listed here so the IDE keeps recognising the board across a reset.
# 0x1209 is pid.codes, the vendor ID shared by open source hardware projects,
# and these three product IDs were free in its registry when they were picked.
#   0xB750  bootloader, CDC + HID
#   0xB751  bootloader, CDC + HID + UF2 mass storage (after a reset double-tap)
#   0xB752  the sketch
BARAM.vid.0=0x1209
BARAM.pid.0=0xB750
BARAM.vid.1=0x1209
BARAM.pid.1=0xB751
BARAM.vid.2=0x1209
BARAM.pid.2=0xB752

# What the sketch itself enumerates as. platform.txt defaults build.vid/build.pid
# to vid.0/pid.0, which would make the sketch impersonate the bootloader.
BARAM.build.vid=0x1209
BARAM.build.pid=0xB752
BARAM.build.usb_manufacturer="BARAM"
BARAM.build.usb_product="WEACT-H750-MINI"

# WEACT-H750-MINI
# STM32H750VBT6, custom bootloader in internal flash, application runs XiP from
# external QSPI flash (W25Q64 8MB @ 0x90000000).
#   0x90000000 4K  TAG sector   (written by the bootloader, NOT part of the image)
#   0x90001000 1K  app vectors  <- VTOR
#   0x90001400 1K  app firm_ver_t
#   0x90001800     app code
# VECT_TAB_BASE_ADDRESS is used instead of build.flash_offset because the core does
# SCB->VTOR = VECT_TAB_BASE_ADDRESS | VECT_TAB_OFFSET (an OR, not an addition).
BARAM.menu.pnum.WEACT_H750_MINI=WEACT-H750-MINI
BARAM.menu.pnum.WEACT_H750_MINI.build.board=WEACT_H750_MINI
BARAM.menu.pnum.WEACT_H750_MINI.build.product_line=STM32H750xx
BARAM.menu.pnum.WEACT_H750_MINI.build.variant=STM32H7xx/WEACT_H750_MINI
BARAM.menu.pnum.WEACT_H750_MINI.build.variant_h=variant_WEACT_H750_MINI.h
BARAM.menu.pnum.WEACT_H750_MINI.build.st_extra_flags=-DCORE_CM7 -D{build.product_line} -DVECT_TAB_BASE_ADDRESS=0x90001000 {build.enable_usb} {build.xSerial}
BARAM.menu.pnum.WEACT_H750_MINI.upload.maximum_size=8382464
BARAM.menu.pnum.WEACT_H750_MINI.upload.maximum_data_size=524288
BARAM.menu.pnum.WEACT_H750_MINI.bootloader.tool=bootburn
BARAM.menu.pnum.WEACT_H750_MINI.build.bootloader_name=weact_h750_mini
BARAM.menu.pnum.WEACT_H750_MINI.build.qspi_cfg=weact_h750_qspi.cfg
BARAM.menu.pnum.WEACT_H750_MINI.debug.svd_file={runtime.tools.STM32_SVD.path}/svd/STM32H7xx/STM32H750.svd

# Upload methods

# --- BARAM -----------------------------------------------------------------------

# Drag-and-drop over the bootloader's UF2 mass storage volume.
# Requires the bootloader's USB stack, which is not implemented yet.
# The default: no debug probe, just the USB cable. arduino-cli performs the
# 1200 bps touch first, which the sketch answers by rebooting into the
# bootloader, then waits for the bootloader's port to appear.
BARAM.menu.upload_method.usbMethod=Bootloader USB (CDC)
BARAM.menu.upload_method.usbMethod.upload.protocol=serial
BARAM.menu.upload_method.usbMethod.upload.tool=baram_boot
BARAM.menu.upload_method.usbMethod.upload.use_1200bps_touch=false
BARAM.menu.upload_method.usbMethod.upload.wait_for_upload_port=false

# Writes the application into the external QSPI flash over SWD using OpenOCD's
# stmqspi driver. The MCU acts as the bridge: OpenOCD drives the QUADSPI
# peripheral registers directly and uploads a small SRAM stub for bulk writes.
# Drag and drop, automated. The bootloader's mass storage volume only comes up
# after a reset double-tap, so this cannot reset the board for you the way the
# CDC method can - double-tap first, then upload. The .uf2 is built either way,
# so it can also just be dragged onto the drive by hand.
BARAM.menu.upload_method.uf2Method=UF2 mass storage (double-tap first)
BARAM.menu.upload_method.uf2Method.upload.protocol=uf2
BARAM.menu.upload_method.uf2Method.upload.tool=baram_uf2

BARAM.menu.upload_method.qspiMethod=OpenOCD QSPI (SWD)
BARAM.menu.upload_method.qspiMethod.upload.protocol=stlink
BARAM.menu.upload_method.qspiMethod.upload.tool=openocd_qspi
