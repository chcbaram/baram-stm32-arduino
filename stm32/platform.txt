# Uploader tools
# -------------------
#
# The application lives in the external QSPI flash, not in internal flash, so
# none of the upstream STM32 upload tools apply: STM32CubeProgrammer cannot
# reach a QSPI device (and is not bundled - STM32Tools only ships a wrapper
# script that looks for a separately installed copy), and the mass storage and
# serial bootloader paths target internal flash.

# Write the sketch into the external QSPI flash over SWD.
#
# OpenOCD does not talk to the flash chip directly. It drives the MCU's QUADSPI
# peripheral over SWD - reading the JEDEC ID, issuing erase and page program
# commands - and uploads a small stub into SRAM for bulk writes. The MCU is the
# bridge, so QUADSPI must be clocked and its pins in alternate function mode
# before this works; weact_h750_qspi.cfg takes care of that.
tools.openocd_qspi.path={openocd_dir}
tools.openocd_qspi.cmd=bin/openocd
tools.openocd_qspi.cmd.windows=bin/openocd.exe
tools.openocd_qspi.upload.params.verbose=-d2
tools.openocd_qspi.upload.params.quiet=-d0
tools.openocd_qspi.upload.pattern="{path}/{cmd}" {upload.verbose} -f interface/{upload.protocol}.cfg -f target/{openocd.target}.cfg -f "{runtime.platform.path}/debugger/{build.qspi_cfg}" -c "program {{build.path}/{build.project_name}.elf} verify reset exit"

# Write the sketch through the board's own bootloader over USB.
#
# The bootloader enumerates as CDC + HID (VID 0xCAFE, PID 0xB010; 0xB011 when
# the UF2 mass storage volume is also present) and accepts a packet protocol on
# the HID interface. download.py speaks it.
# baramdl is a self-contained binary rather than a script: the Arduino IDE
# bundles neither Python nor a JRE, and arduino-cli simply runs upload.pattern
# as a subprocess.
tools.baram_boot.path={baramdl_path}
tools.baram_boot.cmd={baramdl_cmd}
tools.baram_boot.upload.params.verbose=-v
tools.baram_boot.upload.params.quiet=
tools.baram_boot.upload.pattern="{path}/{cmd}" {upload.verbose} --port {serial.port} --write "{build.path}/{build.project_name}.bin"

# Copy the .uf2 onto the bootloader's mass storage volume, which is the same
# thing as dragging the file onto the drive. The volume only appears after a
# reset double-tap, so unlike the CDC path this one is not automatic.
tools.baram_uf2.path={baramdl_path}
tools.baram_uf2.cmd={baramdl_cmd}
tools.baram_uf2.upload.params.verbose=
tools.baram_uf2.upload.params.quiet=
tools.baram_uf2.upload.pattern="{path}/{cmd}" --uf2-copy "{build.path}/{build.project_name}.uf2"
